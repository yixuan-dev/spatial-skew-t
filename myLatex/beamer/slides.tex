\section{Motivation}
% =====================================================================
\begin{frame}{Motivation}
  \begin{itemize}
    \item \textbf{Threshold exceedance} events in air pollution (e.g., ozone) are spatially dependent and heavy-tailed.
    \item Gaussian processes cannot jointly capture \textbf{skewness} and \textbf{tail dependence}.
    \item This work extends the spatio-temporal \skewt process with:
      \begin{enumerate}
        \item AR(2) temporal priors, for richer temporal dynamics;
        \item Multi-resolution spline basis functions, for non-stationary spatial structure.
      \end{enumerate}
  \end{itemize}

  \begin{block}{Goal}
    Quantify the predictive gain of these extensions for \textbf{threshold exceedance prediction} (CRPS, Brier score).
  \end{block}
\end{frame}

\begin{frame}{Related Work}
  \begin{itemize}
    \item \skewt processes and tail-dependence models.
    \item Temporal priors in spatio-temporal models.
    \item Basis-function expansions and autoFRK.
  \end{itemize}
  \vfill

\end{frame}

\section{Methods}
% =====================================================================
\begin{frame}{Model Setup}
  Let $Y(\bs, t)$ be the observation at location $\bs$ and time $t$:
  \begin{equation}
    Y(\bs, t) = \bx(\bs, t)^\top \bbeta + \bB(\bs)^\top \bomega_t + \varepsilon(\bs, t),
  \end{equation}
  where
  \begin{itemize}
    \item $\bB(\bs)$: multi-resolution spline basis functions;
    \item $\bomega_t$: temporal coefficients with an AR(2) prior
      \[
        \bomega_t = \phi_1 \bomega_{t-1} + \phi_2 \bomega_{t-2} + \bfe_t;
      \]
    \item $\varepsilon(\bs, t)$: \skewt error with skewness parameter $\blambda$.
  \end{itemize}
\end{frame}

\begin{frame}{Inference \& Evaluation}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Bayesian inference}
      \begin{itemize}
        \item MCMC sampling.
        \item Identifiability check for $\phi_\tau$.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Predictive scoring}
      \begin{itemize}
        \item CRPS.
        \item Brier score (threshold exceedance).
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Application}
% =====================================================================
\begin{frame}{Ozone Data Analysis}
  \begin{itemize}
    \item Data source and preprocessing.
    \item After removing the autoFRK spatial structure, residuals become \textbf{heavier-tailed} ($\nu$: 15.7 $\to$ 3.8).
    \item Heavy tails are a feature of the data, not a lack of fit.
  \end{itemize}

\section{Conclusion}
% =====================================================================
\begin{frame}{Conclusion \& Contributions}
  \begin{enumerate}
    \item AR(2) temporal priors yield significant predictive gains in the near-unit regime.
    \item Multi-resolution spline bases capture non-stationary spatial structure.
    \item $\phi_\tau$ is identifiable: the posterior concentrates well beyond the prior.
  \end{enumerate}
  \vfill
  \begin{block}{Future Work}
    Higher-dimensional space-time, more threshold settings, and real-time prediction.
  \end{block}
\end{frame}
