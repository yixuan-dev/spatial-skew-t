% ---- p3: Problem background ----
\begin{frame}{Problem background}
  \begin{itemize}
    \item Monitoring networks: limited coverage $\to$ statistical
          extrapolation, with quantified confidence
    \item Monitoring and compliance
    \item Target: $\Pr(Y > L)$ at \alert{unmonitored} sites ---
          the upper tail, not the mean field
  \end{itemize}
  \bigskip
  \begin{center}
    \begin{tabular}{@{}lccc@{}}
      \toprule
      Ozone NAAQS (ppb) & 1997 & 2008 & 2015 (current) \\
      \midrule
      Annual 4th-highest daily max 8-hr avg., 3-year mean & 80 & 75 & 70 \\
      \bottomrule
    \end{tabular}
  \end{center}
\end{frame}

% ---- p4: Why standard methods fall short ----
\begin{frame}{Why standard spatial methods fall short}
  \begin{enumerate}
    \item \textbf{Margin} --- Gaussian: light-tailed, symmetric;
          ozone maxima: \alert{heavy-tailed, right-skewed}
    \item \textbf{Extremal Dependence} \parencite{Coles1999}:
  \end{enumerate}
  \begin{equation*}
    \chi(h) = \lim_{c \to c^*} \Pr[\,Y(\bs + h) > c \mid Y(\bs) > c\,],
    \qquad \text{Gaussian: } \chi(h) = 0 \ \ \forall h
  \end{equation*}
  \bigskip
  \begin{center}
    heavy tails $+$ asymmetry $+$ asymptotic dependence
    $\;\Rightarrow\;$ \alert{skew-$t$ process}
  \end{center}
\end{frame}

% ---- p5: Why not EVT / max-stable ----
\begin{frame}{Why not max-stable or EVT threshold models}
  \begin{itemize}
    \item Max-stable likelihood: infeasible at $1{,}089$ stations
    \item 75 ppb $\approx$ \alert{$0.92$ sample quantile} --- not the far tail in EVT sense
    \item \alert{Subasymptotic} setting \parencite{huser_advances_2022}:
          joint tail at high but \emph{finite} levels
    \item Skew-$t$ = location-scale mixture of GPs $\in$ this family
  \end{itemize}
\end{frame}

\section{Model}
% =====================================================================

% ---- p6a: Baseline STP, model and parameters ----
\begin{frame}{Baseline STP: model and parameters}
  \textcite{morris_space-time_2017}:
  \begin{equation*}
    Y_t(\bs) \;=\;
    \underbrace{\bX_t(\bs)^\top \bbeta}_{\text{regression mean}}
    \;+\;
    \underbrace{\lambda\, \sigma_t(\bs)\, |z_t(\bs)|}_{\text{skewness term}}
    \;+\;
    \underbrace{\sigma_t(\bs)\, v_t(\bs)}_{\text{spatial process}}
  \end{equation*}
  \bigskip
  \begin{itemize}
    \item $\lambda \in \calR$: skewness $\cdot$ $z_t(\bs) \sim N(0, 1)$,
          $|z_t|$ half-normal
    \item $\sigma^2_t(\bs) \sim \mathrm{IG}(a/2,\, b/2)$: scale mixing
          $\to$ heavy tails ($a$ = df)
    \item $v_t(\bs)$: standard GP, \Matern correlation
          ($\varphi$ range, $\nu$ smoothness, $\eta$ spatial proportion)
    \item Marginalize $|z_t|$, $\sigma_t$ $\;\Rightarrow\;$
          \alert{skew-$t$ margin} $+$ asymptotic dependence
  \end{itemize}
\end{frame}

% ---- p6b: Baseline STP, Morris's extensions ----
\begin{frame}{Baseline STP: Morris's extensions}
  \begin{enumerate}
    \item \textbf{Random partition}, $K$ knots redrawn daily:
      \begin{equation*}
        P_{tk} = \{\bs : k = \argmin_\ell \|\bs - \bw_{t\ell}\|\},
        \qquad z_t(\bs) = z_{tk}, \quad \sigma^2_t(\bs) = \sigma^2_{tk}
      \end{equation*}
      $\to$ \alert{local} asymptotic dependence
      \medskip
    \item \textbf{Censored likelihood}:
      $\widetilde{Y}_t(\bs) = \max\{Y_t(\bs),\, T\}$, $T \le L$ ---
      upper-tail focus; below $T$: imputed in MCMC
      \medskip
    \item \textbf{Temporal dependence}: PIT $\to$ Gaussian scale $\to$
      stationary \alert{AR(1)}; margins preserved at every $t$
  \end{enumerate}
\end{frame}

% ---- p8: Extension 1, standardized AR(2) ----
\begin{frame}{Extension 1: a standardized AR(2) prior}
  On the PIT-transformed Gaussian scale:
  \begin{equation*}
    Y_t = \phi_1 Y_{t-1} + \phi_2 Y_{t-2} + \epsilon_t,
    \qquad (\phi_1, \phi_2) \in \text{stationarity triangle}
  \end{equation*}
  Inverse PIT needs $\gamma_0 = \mathrm{Var}(Y_t) = 1$
  $\;\Rightarrow\;$ $\sigma^2_\epsilon$ \alert{not free} (Yule--Walker):
  \begin{equation*}
    \gamma_1 = \frac{\phi_1}{1 - \phi_2}, \qquad
    \gamma_2 = \phi_1 \gamma_1 + \phi_2, \qquad
    \sigma^2_\epsilon = 1 - \phi_1 \gamma_1 - \phi_2 \gamma_2
  \end{equation*}
  \bigskip
  \begin{itemize}
    \item Initial pair $(Y_1, Y_2)$: stationary bivariate Gaussian,
          correlation $\gamma_1$
    \item Identical margins across i.i.d.\ / AR(1) / AR(2)
  \end{itemize}
\end{frame}

% ---- p9: Extension 2, MRTS ----
\begin{frame}{Extension 2: MRTS basis functions in the mean}
  MRTS \parencite{tzeng_resolution_2018}:
  \begin{itemize}
    \item Ordered \alert{coarse $\to$ fine}: $f_1 = 1$; $f_2, f_3$ =
          coordinates; $k \ge 4$: eigenvectors of the TPS kernel,
          eigenvalues decreasing
    \item Closed form at \emph{new} locations $\to$ predictive covariate
  \end{itemize}
  \bigskip
  \begin{equation*}
    \bX^*_t(\bs) = \big[\bff(\bs)^\top, \bX_t(\bs)^\top\big]^\top,
    \qquad
    Y_t(\bs) = \bX^*_t(\bs)^\top \bbeta^* + \lambda\,\sigma_t(\bs)\,|z_t(\bs)|
               + \sigma_t(\bs)\, v_t(\bs)
  \end{equation*}
  \bigskip
  \begin{itemize}
    \item Claim: non-stationary \alert{mean} only ---
          \Matern keeps small-scale dependence
  \end{itemize}
\end{frame}

% ---- p10: Hierarchical model ----
\begin{frame}{Hierarchical model and inference}
  Three stages \parencite{berliner_hierarchical_1996}:
  \begin{description}
    \item[Stage 1] censored data:
      $\widetilde{Y}_t(\bs_i) = \max\{Y_t(\bs_i), T\}$
    \item[Stage 2] Gaussian field $\cdot$ partition $\cdot$ latent AR(2)
      dynamics
    \item[Stage 3] priors; $(\phi_1, \phi_2) \sim$ Unif(stationarity triangle)
  \end{description}
  \begin{equation*}
    \big[\bY_{\mathrm{cens}}, \calZ, \btheta \mid \widetilde{\bY}\big]
    \;\propto\;
    \prod_{t=1}^{n_t}
    \big[\widetilde{\bY}_t \mid \bY_t\big]
    \big[\bY_t \mid \calZ_t, \btheta\big]\;
    \big[\calZ_{1:n_t} \mid \btheta\big]\,
    \big[\btheta\big]
  \end{equation*}
  \bigskip
  \begin{itemize}
    \item Gibbs-within-MH; AR(2) full conditionals: appendix
    \item $K$ fixed $=$ \alert{candidate model index}, chosen by held-out Brier
  \end{itemize}
\end{frame}

% ---- B1: lambda-z reflection ridge & screening ----
\begin{frame}{Backup: $\lambda$--$z$ reflection ridge and forecast screening}
  \begin{itemize}
    \item A reflection-type non-identifiability links $\lambda$ and $z$:
          short-record fits can land on the \alert{reflected ridge}, where the
          skewness parameter and the intercept fail to separate and the
          predictive spread inflates.
    \item The screening rule is public and stated with the forecast protocol:
          fits whose $\lambda$ and intercept fail to separate, or whose
          predictive spread is unbounded, are removed \emph{before} scoring.
    \item Root fix --- a sign-anchored reparameterization of $z$ --- is listed
          as future work.
  \end{itemize}
\end{frame}

% ---- B2: AR(2) derivations ----
\begin{frame}{Backup: AR(2) stationary moments and full conditionals}
  \begin{itemize}
    \item Yule--Walker under $\gamma_0 = 1$:
          $\gamma_h = \phi_1 \gamma_{h-1} + \phi_2 \gamma_{h-2}$ for
          $h \ge 1$ gives $\gamma_1 = \phi_1 / (1 - \phi_2)$,
          $\gamma_2 = \phi_1 \gamma_1 + \phi_2$, and
          $\sigma^2_\epsilon = 1 - \phi_1 \gamma_1 - \phi_2 \gamma_2$.
    \item Gibbs-within-MH: the full conditional of $Y_t$ carries \alert{three}
          prior factors,
          \begin{equation*}
            \pi(Y_t \mid Y_{-t}, \bY) \propto L(\bY \mid Y_t)\,
            \underbrace{p(Y_t \mid Y_{t-1}, Y_{t-2})}_{\text{self}}\,
            \underbrace{p(Y_{t+1} \mid Y_t, Y_{t-1})}_{\text{lag-1}}\,
            \underbrace{p(Y_{t+2} \mid Y_{t+1}, Y_t)}_{\text{lag-2}};
          \end{equation*}
          omitting the lag-2 factor distorts the stationary distribution.
    \item Head of the series ($t = 1, 2$): the stationary initial-pair density
          replaces the self factor.  Proposals: symmetric Gaussian random walk
          tuned to $0.3$--$0.4$ acceptance; $(\phi_1, \phi_2)$ updated jointly
          under the uniform prior on the stationarity triangle.
  \end{itemize}
\end{frame}

% ---- B3: MRTS second-moment design details ----
\begin{frame}{Backup: second-moment design --- ``inert'' is not ``no signal''}
  \begin{itemize}
    \item The random effect superposes two fixed cosine fields with
          daily-redrawn Gaussian weights:
          \begin{equation*}
            u_t(\bs) = \sum_{j=1}^{2} \xi_{tj}\, g_j(\bs), \quad
            \xi_{tj} \iid N(0, \tau_j^2), \quad
            (\tau_1, \tau_2) = (5, 3),
          \end{equation*}
          giving the location-dependent covariance
          $\mathrm{Cov}\{u_t(\bs), u_t(\bs')\}
           = \sum_j \tau_j^2 g_j(\bs) g_j(\bs')$.
    \item Zero time-average $\Rightarrow$ the pooled coefficient has
          \alert{no first-moment structure to recover}, while $u_t$ still
          contributes substantial non-stationary variance: structurally out of
          reach for a fixed-effect mean, not absent signal.
    \item The two nonlinear-mean surfaces were confirmed by an \alert{oracle
          calculation before any fitting} to leave genuine Brier headroom.
  \end{itemize}
\end{frame}

% ---- B4: censoring pairing & MCMC ----
\begin{frame}{Backup: censoring pairing and MCMC settings}
  \begin{itemize}
    \item Why symmetric-$t$ with censoring: once the lower tail is censored,
          the skewness parameter is only \alert{weakly identified}, so
          censored settings pair with a symmetric-$t$ marginal, following
          \textcite{morris_space-time_2017}.
    \item Thresholds: the censor level $T = 75$ ppb sits below the era
          standard $L = 80$ ppb ($T \le L$), and every evaluation level lies
          above $T$; the degrees of freedom $a$ control the tail thickness.
    \item Burn-in $25{,}000$ of $30{,}000$ ($\approx 83\%$): mixing is slowed
          by censoring imputation and the latent volume; traces were
          inspected, and the long burn-in is a conservative choice.
  \end{itemize}
\end{frame}
